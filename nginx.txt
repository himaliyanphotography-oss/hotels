    listen 80;
    server_name Spellmount .in www.Spellmount .in;
    return 301 https://Spellmount .in$request_uri;
}

server {
    listen 443 ssl; # Managed by Certbot
    server_name Spellmount .in www.Spellmount .in;

    ssl_certificate /etc/letsencrypt/live/Spellmount .in/fullchain.pem; # Managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/Spellmount .in/privkey.pem; # Managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # Managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # Managed by Certbot

    location / {
        proxy_pass http://localhost:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;

        # Cache static content for 2 days (172800 seconds)
        add_header Cache-Control "max-age=172800, must-revalidate";
        add_header Expires $date_gmt;

        # Allow caching by browsers
        expires 2d;
    }

    # Redirect www to non-www
    if ($host = www.Spellmount .in) {
        return 301 https://Spellmount .in$request_uri;
    }
}
